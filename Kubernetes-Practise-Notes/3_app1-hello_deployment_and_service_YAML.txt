1.0--> create app.yaml

   
			apiVersion: apps/v1
			kind: Deployment
			metadata:
			  name: app1-hello
			spec:
			  replicas: 1
			  selector:
				matchLabels:
				  app: app1-hello
			  template:
				metadata:
				  labels:
					app: app1-hello
				spec:
				  containers:
					- name: app1-hello
					  image: mishalpandey/hello-spring:1.0
					  env:
						- name: ENV_MESSAGE
						  valueFrom:
							configMapKeyRef:
							  name: app-config
							  key: ENV_MESSAGE
						- name: ENV_VALUE
						  valueFrom:
							secretKeyRef:
							  name: app-secret
							  key: ENV_VALUE
					  ports:
						- containerPort: 8080
			---
			apiVersion: v1
			kind: Service
			metadata:
			  name: app1-hello
			spec:
			  selector:
				app: app1-hello
			  type: NodePort
			  ports:
				- port: 80
				  targetPort: 8080
				  nodePort: 32200
				  
				  
1.1--> create config.yaml


			apiVersion: v1
			kind: ConfigMap
			metadata:
			  name: app-config
			data:
			  ENV_MESSAGE: "ENV VALUE is in secret from Kubernetes ConfigMap!"


			#data:
			#  ENV_MESSAGE: "ENV VALUE is in secret from Kubernetes ConfigMap!"
			# it will work here because in the configMap not needed base64 encoding value, so we can use data as well here and stringData.
			
			

1.2--> create secret.yaml

			apiVersion: v1
			kind: Secret
			metadata:
			  name: app-secret
			type: Opaque
			stringData:
			  ENV_VALUE: "Hello I am coming from secret in Kubernetes Secret!"

			#data:
			#  ENV_VALUE: "Hello I am coming from secret in Kubernetes Secret!"
			# it will not work because the value is not base64 encoded, so we use stringData instead of data.
			
			

1.3--> kubectl apply -f secret.yaml -f config.yaml -f app.yaml

		deployment.apps/app1-hello created
		service/app1-hello created
		config/app1-hello created
