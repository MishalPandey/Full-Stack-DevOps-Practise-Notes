				metadata:

3.0--> create service.yaml

			apiVersion: v1
			kind: Service
			metadata:
			  name: myapp
			spec:
			  selector:
				app: myapp
			  type: NodePort
			  ports:
				- port: 80                                     //This is service port traffic comes from http://localhost:32000/ here 
				  targetPort: 80                               //Then this service port is mapped to pods/Container port that is targetPort
				  nodePort: 32000                              //This is browers port http://localhost:32000/ locally we can access via this.

				type: Opaque
				data:
